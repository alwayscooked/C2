<script>
    async function getobj(){
        var host = document.getElementById('host').value
        var tactic = document.getElementById('tactic').value
        const request = new Request('https://###CHANGE_FOR_IP###/get_results?host='+host+'&tactic='+tactic)
        const response = await fetch(request)
        const res = await response.json()
        let div_res = document.getElementById("res")
        div_res.innerHTML = ''
        if(tactic=='T1113' || tactic=='T1125'){
            for(let i = 0; i<res['data'].length; i++){
                div_res.innerHTML+='<p><a href="./'+res['data'][i]+'">'+res['data'][i].split("/").pop()+'</a></p>'
            }
        }else if(tactic=='T1123'){
            for(let i = 0; i<res['data'].length; i++){
                div_res.innerHTML+='<p><audio controls src="./'+res['data'][i]+'" type="audio/wav"></audio></p>'
            }
        }else{
            let textarea = document.createElement('textarea')
            textarea.name="aval_hosts" 
            textarea.id="ahosts" 
            textarea.readOnly=true
            textarea.rows="18" 
            textarea.cols="80"
            textarea.value = res['data']
            div_res.append(textarea)
        }
        
    }
</script>
